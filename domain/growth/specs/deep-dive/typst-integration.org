* Overview

Live Typst editing with branded PDF output. Two halves: an editor with
tree-sitter syntax highlighting, and a WebEngineView preview rendering via
typst-ts WASM. Connected by click-to-code source mapping.

The system serves two writers: a human in the editor, and the agent pushing
text through =bridge.send()=. Both paths converge on the same TextEdit, same
parse, same preview pipeline.

* Architecture

** Editor Component

Qt-TreeSitterHighlighter (C++ library, Qt6) wraps tree-sitter grammars and
applies syntax highlighting to a QML =TextEdit=.

- Grammar: =uben0/tree-sitter-typst= — supports markup, code, and math modes
  with indentation queries and folding ranges.
- Highlighting: The C++ highlighter attaches to =TextEdit.textDocument=,
  incrementally re-highlights on each edit. No full reparse on keystroke.
- QML surface: =TextEdit= with =font.family: "Aspekta"=, line numbers gutter,
  and bracket matching derived from tree-sitter node queries.

#+begin_src qml
// TypstEditor.qml — editor pane

import QtQuick.Controls 2.15

        document: textEdit.textDocument
        grammar: "typst"  // resolved to tree-sitter-typst.wasm or .so

        contentHeight: textEdit.contentHeight

        textEdit.cursorPosition = pos
        textEdit.forceActiveFocus()
    }

** Preview Component

=WebEngineView= loads a minimal HTML page that bootstraps the typst-ts WASM
compiler and renderer. On each text change (debounced), the main content is
sent into the WASM compiler which produces SVG output injected into the DOM.

Key packages:
| Package                                | Role                          |
|----------------------------------------+-------------------------------|
| =@myriaddreamin/typst-ts-web-compiler= | WASM module — Typst → vector IR |
| =@myriaddreamin/typst-ts-renderer=     | Vector IR → SVG DOM           |
| =@myriaddreamin/typst.ts=              | High-level API coordinating both |

typst-ts uses Reflexo vector IR as an intermediate representation. This enables
incremental compilation: only changed pages re-render, unchanged pages reuse
cached IR. This is what makes 200ms debounce feel instant for most edits.

#+begin_src html
<!-- typst-preview.html — loaded into WebEngineView -->
<!DOCTYPE html>
<html>
<head>

    body { margin: 0; background: #2b2b2b; display: flex; justify-content: center; }

<body>
  <div id="container"></div>
  <script type="module">
    import { createTypstCompiler } from '@myriaddreamin/typst.ts';

    const compiler = await createTypstCompiler();
    await compiler.init();

    // Load ORA brand assets into virtual filesystem
    // (populated by loadVirtualFS call from QML side)
    window.$typst = {
      compiler,
      virtualFS: new Map(),

      async loadFile(path, content) {
        this.virtualFS.set(path, content);
        compiler.addSource(path, content);
      },

      async loadFont(fontData) {
        compiler.addFontData(fontData);
      },

      async svg(opts) {
        compiler.addSource('/main.typ', opts.mainContent);
        const doc = await compiler.compile({ mainFilePath: '/main.typ' });
        if (doc.diagnostics?.length) {
          window.bridge?.send({ type: 'typst:diagnostics', diagnostics: doc.diagnostics });
        }
        if (doc.result) {
          const svg = await compiler.renderToSvg(doc.result);
          document.getElementById('container').innerHTML = svg;
          attachSourceMapHandlers();
        }
      }
    };

    function attachSourceMapHandlers() {
      document.querySelectorAll('[data-source-line]').forEach(el => {
        el.addEventListener('click', (e) => {
          e.stopPropagation();
          const line = parseInt(el.getAttribute('data-source-line'));
          const col = parseInt(el.getAttribute('data-source-column') || '0');
          // Send to QML bridge — editor jumps to source
          window.bridge?.send({ type: 'typst:goto-source', line, column: col });
        });
      });
    }

    // Expose render trigger for QML evaluate() calls
    window.renderTypst = (content) => $typst.svg({ mainContent: content });
  </script>
</body>
</html>

// TypstPreview.qml — preview pane

import QtWebEngine 1.10

Item {
    id: previewPane
    signal gotoSource(int line, int column)

    WebEngineView {
        id: webView
        anchors.fill: parent
        url: "qrc:/typst-preview.html"

            if (info.status === WebEngineLoadingInfo.LoadingSucceededStatus) {
                previewPane.initVirtualFS()
            }
        }

        // Receive messages from page JS
        WebChannel {
            id: channel
        }
    }

    Connections {
        target: bridge
        function onMessageReceived(payload) {
            if (payload.type === "typst:goto-source") {
                previewPane.gotoSource(payload.line, payload.column)
            }
        }
    }

    function render(content: string) {
        webView.runJavaScript(`window.renderTypst(${JSON.stringify(content)})`)
    }

        // Called after WebEngineView loads — populates brand assets
        // Bridge provides file list from domain/growth/templates/typst/

    function loadFile(path: string, content: string) {
        webView.runJavaScript(
            `window.$typst.loadFile(${JSON.stringify(path)}, ${JSON.stringify(content)})`
        )
    }

    function loadFont(base64Data: string) {
        webView.runJavaScript(
            `window.$typst.loadFont(Uint8Array.from(atob(${JSON.stringify(base64Data)}), c => c.charCodeAt(0)))`

** Click-to-Code

typst-ts source mapping works through the vector IR layer. Compiled SVG
elements carry =data-source-line= and =data-source-column= attributes when
source mapping is enabled in the compiler options.

Flow:
1. User clicks an element in the preview WebEngineView.
2. JS click handler reads =data-source-line= / =data-source-column=.
3. JS calls =window.bridge.send({ type: 'typst:goto-source', line, column })=.
4. QML =Connections= handler on bridge receives the message.
5. Emits =gotoSource(line, column)= signal.
6. Parent =SplitView= calls =editor.jumpToLine(line)=.

Reverse direction (cursor-to-preview scroll) is a stretch goal — requires
mapping editor cursor position back to SVG element IDs.

** Font Loading

Fonts must be loaded into the WASM compiler as binary data. On project open,
the TypeScript side reads font files and sends them to the preview:

#+begin_src typescript
// packages/coding-agent/src/domains/growth/typst-fonts.ts
import { readFile } from 'fs/promises';
import { resolve } from 'path';

const FONT_DIR = resolve(brandingPath, 'fonts');

const REQUIRED_FONTS = [
  'Aspekta-300.woff2',
  'Aspekta-400.woff2',
  'Aspekta-500.woff2',
  'Aspekta-700.woff2',
  'Aspekta-950.woff2',
  'FunnelDisplay-Regular.woff2',
];

export async function loadBrandFonts(
  sendToPreview: (msg: { type: string; base64: string; name: string }) => void
): Promise<void> {
  for (const font of REQUIRED_FONTS) {
    const data = await readFile(resolve(FONT_DIR, font));
    sendToPreview({
      type: 'typst:load-font',
      base64: data.toString('base64'),
      name: font,
    });
  }
}
#+end_src

** Virtual Filesystem Population

#+begin_src typescript
// packages/coding-agent/src/domains/growth/typst-vfs.ts
import { readFile, readdir } from 'fs/promises';
import { resolve, relative } from 'path';

const BRAND_ROOT = '~/code/ora/ora-branding/typst/ora-brand';

export async function populateVirtualFS(
  brandRoot: string,
  sendToPreview: (msg: { type: string; path: string; content: string }) => void
): Promise<void> {
  const files = await collectTypFiles(brandRoot);
  for (const absPath of files) {
    const relPath = 'ora-brand/' + relative(brandRoot, absPath);
    const content = await readFile(absPath, 'utf-8');
    sendToPreview({ type: 'typst:load-file', path: relPath, content });
  }
}

async function collectTypFiles(dir: string): Promise<string[]> {
  const entries = await readdir(dir, { withFileTypes: true, recursive: true });
  return entries
    .filter(e => e.isFile() && e.name.endsWith('.typ'))
    .map(e => resolve(e.parentPath, e.name));
}

** Variable Injection

The agent populates variables by writing a =data/variables.json= file into the
virtual filesystem before compilation. This avoids string interpolation in Typst
source — the template reads data via =#json()= which is a native Typst function.

#+begin_src typescript
// Agent populating template variables
async function renderTemplate(
  templateName: string,
  variables: Record<string, unknown>,
  sendToPreview: (msg: { type: string; path: string; content: string }) => void
): Promise<void> {
  // Load template source
  const templatePath = `domain/growth/templates/typst/${templateName}.typ`;
  const source = await readFile(templatePath, 'utf-8');

  // Inject variables as JSON data file
  sendToPreview({
    type: 'typst:load-file',
    path: 'data/variables.json',
    content: JSON.stringify(variables),
  });

  // Set as main document — triggers compile + preview
  sendToPreview({
    type: 'typst:set-content',
    content: source,
  });
}
#+end_src

* Compilation Pipeline

** Development: WASM Preview

Used during editing for instant feedback.

       │

Limitations of WASM preview:
- Font rendering may differ from CLI (subset vs full font files)
- Some Typst packages may not work in WASM (those requiring filesystem access)
- Large documents (50+ pages) may lag — mitigated by page-level incremental rendering

** Export: CLI Compilation

For final PDF output, shell out to =typst compile= which has full system font
access and native performance.

#+begin_src typescript
// packages/coding-agent/src/domains/growth/typst-export.ts
import { execFile } from 'child_process';
import { writeFile, mkdir } from 'fs/promises';
import { resolve, dirname } from 'path';

export interface TypstExportOptions {
  source: string;           // main .typ file path
  output: string;           // output .pdf path
  fontPaths?: string[];     // additional font directories
  root?: string;            // typst project root (for imports)
}

export async function exportPDF(opts: TypstExportOptions): Promise<string> {
  const args = ['compile', opts.source, opts.output];

  if (opts.root) {
    args.push('--root', opts.root);
  }

  for (const fp of opts.fontPaths ?? []) {
    args.push('--font-path', fp);
  }

  await mkdir(dirname(opts.output), { recursive: true });

  return new Promise((resolve, reject) => {
    execFile('typst', args, (error, stdout, stderr) => {
      if (error) {
        reject(new Error(`typst compile failed: ${stderr}`));
      } else {
        resolve(opts.output);
      }
    });
  });
}
#+end_src

Usage from the agent:

- Error count badge in the preview pane tab

** WASM Failures

If the WASM compiler fails to initialize (browser restrictions, memory limits):
- Fall back to CLI compilation with =typst compile --format svg=
- Preview updates slower (file write → compile → read SVG → inject) but works
- Log the WASM failure for debugging

* Performance Considerations

| Concern                  | Mitigation                                         |
|--------------------------+----------------------------------------------------|
| Large documents          | Page-level incremental rendering via Reflexo IR     |
| Frequent edits           | 200ms debounce before compile                       |
| Font loading             | Load fonts once on project open, cache in WASM      |
| Virtual FS size          | Lazy-load imports on first reference                |
| WebEngineView memory     | Single WebEngineView instance, reuse across docs    |
| Tree-sitter parse time   | Incremental parsing — O(edit size) not O(file size) |

For documents exceeding ~100 pages, the preview can switch to rendering only
the current page and its neighbors (viewport-aware rendering), with full
document compilation deferred to export.
