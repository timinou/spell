** 2. Fix CLI config-dir fallback
File: =packages/coding-agent/src/commands/server.ts=

Current =resolveDefaultConfigDir()= always returns =cwd/.spell/= without checking whether it exists. The planned contract says: try =cwd/.spell/server.kdl= first; if not found, fall back to =~/.spell/=.

Replace:
#+begin_src typescript
import * as os from "node:os";
import * as fs from "node:fs";

function resolveDefaultConfigDir(): string {
  const cwdConfig = path.join(process.cwd(), ".spell");
  try {
    fs.accessSync(path.join(cwdConfig, "server.kdl"));
    return cwdConfig;
  } catch {
    return path.join(os.homedir(), ".spell");
  }
}
#+end_src
