* Problem

When a throttled render is pending (=#renderRequested = true=, throttle timer running), calling =requestRender(force=true)= clears the timer but then hits the =if (this.#renderRequested) return= guard — the force render is never scheduled. The cleared timer means NO render fires at all until the next =requestRender()= call.

This causes screen corruption when resize occurs during subagent execution: niri overview toggle fires =requestRender(true)= which gets swallowed, leaving the screen stale.

** Root cause

=packages/tui/src/tui.ts= lines 625-660:

#+begin_src typescript
requestRender(force = false): void {
    if (force) {
        // ... resets state, clears throttle timer
    }
    if (this.#renderRequested) return; // <-- BUG: blocks force render
    this.#renderRequested = true;
    // ... scheduling
}
#+end_src

The =force= block clears the timer but doesn't reset =#renderRequested=. The guard on the next line returns early.

** Fix

Add =#renderRequested = false= inside the =force= block, before the guard:

#+begin_src typescript
requestRender(force = false): void {
    if (force) {
        this.#previousLines = [];
        this.#previousWidth = -1;
        this.#cursorRow = 0;
        this.#hardwareCursorRow = 0;
        this.#viewportTopRow = 0;
        this.#maxLinesRendered = 0;
        if (this.#throttleTimer) {
            clearTimeout(this.#throttleTimer);
            this.#throttleTimer = undefined;
        }
        this.#renderRequested = false; // <-- FIX: allow force to reschedule
    }
    if (this.#renderRequested) return;
    this.#renderRequested = true;
    // ... rest unchanged
}
#+end_src

** Test (test-first)

Add to =packages/tui/test/render-scheduling.test.ts=:

#+begin_src typescript
it("force=true preempts pending throttle timer", async () => {
    const term = new VirtualTerminal(40, 10);
    const tui = createTUI(term, { minRenderInterval: 50 });
    activeTuis.push(tui);
    const component = new StaticComponent(["initial"]);
    tui.addChild(component);

    tui.start();
    await settleImmediate();
    await term.flush();

    const baselineRenderCount = component.renderCount;

    // Trigger a throttled render (within interval, creates setTimeout)
    component.setLines(["throttled"]);
    tui.requestRender();
    // Now #renderRequested=true, throttle timer pending

    // Force render should preempt
    component.setLines(["forced"]);
    tui.requestRender(true);

    await settleImmediate();
    await term.flush();
    // Force render must have fired, showing "forced"
    expect(containsLine(term, "forced")).toBe(true);
    expect(component.renderCount).toBe(baselineRenderCount + 1);
});
#+end_src
