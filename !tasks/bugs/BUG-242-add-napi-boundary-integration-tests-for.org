* Problem
The code tool's TypeScript test suite mocks =executeCodeBuffer= for all tests. There are no automated JS-level tests that exercise the real Rust→NAPI→TypeScript path for the error-envelope pattern introduced in PLAN-213. The Rust side has unit tests, but a regression in the NAPI bridge, JS wrapper, or result shape could slip through undetected.

An established pattern exists in =packages/natives/test/org-buffer.test.ts= for direct NAPI testing.

* Design Decisions
1. Add =packages/natives/test/code-buffer.test.ts= following the =org-buffer.test.ts= pattern: direct =require= of the =.node= addon, no mocks.
2. Test the error envelope contract: unsupported language, missing command, and a success case.
3. Add a drift-detection test: query the native =languages= command, extract all extensions, assert they match the known set. If the native engine adds a language, this test fails and signals the developer to update =FALLBACK_EXTENSIONS= in =code.ts=.
4. Use =tempfile= (via =node:fs= + =node:os=) for creating test files with unsupported extensions.

import { describe, expect, it } from "bun:test";
import { createRequire } from "node:module";
import * as fs from "node:fs";
import * as os from "node:os";
import * as path from "node:path";

// Load native addon directly, bypassing the embedded-addon import chain
const require_ = createRequire(import.meta.url);
const nativesDir = path.join(import.meta.dir, "..", "native");
const native = require_(path.join(nativesDir, "pi_natives.dev.node")) as {
    executeCodeBuffer(options: Record<string, unknown>): { output: unknown; error: boolean };
};

function executeCodeBuffer(options: Record<string, unknown>): { output: unknown; error: boolean } {
    return native.executeCodeBuffer(options);
}

** Test: unsupported language returns error envelope

#+begin_src typescript
describe("executeCodeBuffer NAPI bridge", () => {
    it("returns error envelope for unsupported language", () => {
        const tmpDir = fs.mkdtempSync(path.join(os.tmpdir(), "code-buffer-test-"));
        const mdFile = path.join(tmpDir, "readme.md");
        fs.writeFileSync(mdFile, "# Hello
");

        try {
            const result = executeCodeBuffer({ command: "read", file: mdFile });
            expect(result.error).toBe(true);
            expect(typeof result.output).toBe("string");
            expect(result.output as string).toContain("language");
        } finally {
            fs.rmSync(tmpDir, { recursive: true, force: true });
        }
    });
#+end_src

** Test: languages command returns extension list

#+begin_src typescript
    it("returns language list via languages command", () => {
        const result = executeCodeBuffer({ command: "languages" });
        expect(result.error).toBe(false);
        const output = result.output as { languages: Array<{ id: string; extensions: string[] }> };
        expect(Array.isArray(output.languages)).toBe(true);
        expect(output.languages.length).toBeGreaterThan(0);

        // Every language entry has a string id and array of extensions
        for (const lang of output.languages) {
            expect(typeof lang.id).toBe("string");
            expect(Array.isArray(lang.extensions)).toBe(true);
            expect(lang.extensions.length).toBeGreaterThan(0);
        }
    });
#+end_src

** Test: native extensions match known fallback set (drift detection)

#+begin_src typescript
    it("native language extensions match FALLBACK_EXTENSIONS in code tool", () => {
        const result = executeCodeBuffer({ command: "languages" });
        expect(result.error).toBe(false);
        const output = result.output as { languages: Array<{ extensions: string[] }> };

        const nativeExtensions = new Set<string>();
        for (const lang of output.languages) {
            for (const ext of lang.extensions) {
                nativeExtensions.add(ext);
            }
        }

        // Must match FALLBACK_EXTENSIONS in packages/coding-agent/src/tools/code.ts
        // If this fails, a language was added/removed in the native engine.
        // Update FALLBACK_EXTENSIONS to match.
        const expected = new Set([
            "ts", "tsx", "js", "jsx", "mjs", "cjs", "mts", "cts",
            "rs",
            "py", "pyi",
            "typ",
            "ex", "exs",
        ]);
        expect(nativeExtensions).toEqual(expected);
    });
});
#+end_src
