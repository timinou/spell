#+begin_src typescript
// BEFORE (drops bufferInvalidationError):
if (this.session.settings.get("edit.previewResolvePolicy") === "auto-apply") {
  return (await applyPendingAction(previewAction, "Auto-applied by edit.previewResolvePolicy"))
    .result as AgentToolResult<AstEditToolDetails>;
}

// AFTER (surfaces degradation):
if (this.session.settings.get("edit.previewResolvePolicy") === "auto-apply") {
  const autoResolution = await applyPendingAction(previewAction, "Auto-applied by edit.previewResolvePolicy");
  const autoResult = autoResolution.result as AgentToolResult<AstEditToolDetails>;
  if (autoResolution.bufferInvalidationError) {
    const existingText = autoResult.content.find(part => part.type === "text")?.text ?? "";
    const warningText = [existingText, autoResolution.bufferInvalidationError].filter(Boolean).join("
");
    autoResult.content = autoResult.content.map(part =>
      part.type === "text" ? { ...part, text: warningText } : part,
    );
    if (autoResult.details) {
      (autoResult.details as AstEditToolDetails).bufferInvalidationError = autoResolution.bufferInvalidationError;
    }
  }
  return autoResult;
}
#+end_src
- This preserves all existing behavior for the happy path (no invalidation error → unchanged result)
- On degradation: appends warning to text content and sets the details field
