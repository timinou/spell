  #+begin_src typescript
  export async function applyPendingAction(action: PendingAction, reason: string): Promise<PendingActionResolution> {
    const resolution = normalizeResolution(await action.apply(reason), {
      mutationState: "applied",
      persisted: true,
      files: action.files,
    });
    if (!resolution) {
      throw new ToolError(`Pending action ${action.label} did not return a result.`);
    }
    if (!resolution.persisted || action.invalidateManagedCodeBuffers !== true) {
      return resolution;
    }
    if (resolution.files.length === 0) {
      return {
        ...resolution,
        bufferInvalidationError:
          "Applied preview persisted to disk, but did not report touched files needed to invalidate managed code buffers.",
      };
    }
    const bufferInvalidationError = await tryInvalidateCodeBuffers(resolution.files);
    return bufferInvalidationError ? { ...resolution, bufferInvalidationError } : resolution;
  }
  #+end_src

* Edge Cases
- Apply callback throws before returning (pre-persist failure): existing behavior preserved — =applyPendingAction= lets it propagate, pending action stays in store
- Apply callback returns bare =AgentToolResult= (no =PendingActionResolution=): =normalizeResolution= fallback assigns =persisted: true= + =files: action.files= → invalidation still attempted correctly via fallback files
- =invalidateManagedCodeBuffers= is false/undefined: invalidation skipped entirely, no error possible
- Multiple files where only one fails close: =tryInvalidateCodeBuffers= returns error on first failure (consistent with current behavior)
- Empty files list from =normalizeResolution= fallback: =action.files= is =undefined= → fallback is =[]= → triggers missing-files warning when opted in
