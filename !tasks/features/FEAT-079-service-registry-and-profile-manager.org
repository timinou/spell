interface ServiceEntry {
  name: string;              // unique key, e.g. 'linkedin', 'google-personal'
  displayName: string;       // human-readable, e.g. 'LinkedIn'
  description: string;       // AI hint, e.g. 'Professional network for research'
  profileStorage: string;    // WebEngine storageName (sanitized)
  domains: string[];         // exact domain matches, e.g. ['linkedin.com', 'www.linkedin.com']
  parentService?: string;    // parent name (shares profileStorage)
  loginUrl?: string;         // URL for agent-driven reconnection
  faviconPath?: string;      // absolute path to extracted favicon PNG
  lastUsed?: string;         // ISO 8601 datetime
  lastValidated?: string;    // ISO 8601 datetime
  status: 'connected' | 'unknown';
}
#+end_src
