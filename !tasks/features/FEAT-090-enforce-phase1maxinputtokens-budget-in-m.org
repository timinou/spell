#+begin_src typescript
// Budget filter: estimate tokens per claim and limit total
let budgetRemaining = config.phase1MaxInputTokens;
const budgetFilteredClaims: Stage1Claim[] = [];
for (const claim of claims) {
    if (budgetFilteredClaims.length > 0 && budgetRemaining <= 0) break;
    let estimatedTokens: number;
    try {
        const size = Bun.file(claim.rolloutPath).size;
        estimatedTokens = Math.ceil(size / 4);
    } catch {
        // Stat failed (file deleted?) — include claim, let runStage1Job handle the error
        estimatedTokens = 0;
    }
    budgetRemaining -= estimatedTokens;
    budgetFilteredClaims.push(claim);
}
if (budgetFilteredClaims.length < claims.length) {
    logger.debug("Memory phase1 budget limited claims", {
        original: claims.length,
        filtered: budgetFilteredClaims.length,
        budget: config.phase1MaxInputTokens,
    });
}
#+end_src

* Edge Cases
- First claim exceeds budget alone: still processed (=budgetFilteredClaims.length > 0= guard). Inner =truncateByApproxTokens= at line ~629 further bounds the actual model input.
- All claims fit budget: no filtering occurs, =budgetFilteredClaims === claims=
- =Bun.file(path).size= failure: treated as 0 tokens, claim included. The downstream =Bun.file(claim.rolloutPath).text()= in =runStage1Job= will fail and hit =markStage1Failed= — correct error path.
- Budget exactly equals first claim: first included, second skipped (=budgetRemaining <= 0= fires)
- Skipped claims in running state: leases expire after =stage1LeaseSeconds= (120s default). Next session startup re-claims them. No retry decrement — correct, these aren't failures.
- =phase1MaxInputTokens= set to 0: first claim still processed (min-one guarantee)
- Negative config value: first claim still processed
