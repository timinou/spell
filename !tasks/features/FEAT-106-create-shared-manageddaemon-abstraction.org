*** Interface
#+begin_src typescript
export interface DaemonConfig {
  /** Human-readable name for logging and postmortem registration (e.g. "emacs-main", "emacs-org") */
  name: string;
  /** Command + args to spawn the daemon. Spawned with --fg-daemon semantics (foreground, no fork). */
  command: string[];
  /** Absolute path to the Unix socket the daemon will create to signal readiness. */
  socketPath: string;
  /** How to gracefully stop the daemon. If omitted, only proc.kill() is used. */
  stopCommand?: string[];
  /** Health check interval in ms (default: 5000). Set 0 to disable. */
  healthIntervalMs?: number;
  /** Maximum time to wait for the socket to appear (default: 30000). */
  startupTimeoutMs?: number;
  /** Spawn environment variables. Merged with process.env. */
  env?: Record<string, string>;
  /** Working directory for the spawned process. */
  cwd?: string;
  /** Pipe stderr and forward lines to logger (default: true). */
  logStderr?: boolean;
  /** Callback fired if socket disappears (daemon crash). */
  onCrash?: () => void;
}

export interface ManagedDaemon {
  /** Absolute path to the Unix socket. */
  readonly socketPath: string;
  /** True when the daemon process is alive and socket exists. */
  isAlive(): boolean;
  /** PID of the daemon process (valid because --fg-daemon, no fork). */
  readonly pid: number;
  /** Graceful stop: send stopCommand, wait briefly, then proc.kill(SIGTERM), then SIGKILL. Removes socket. Deregisters from postmortem. */
  stop(): Promise<void>;
}

/** Spawn a daemon, wait for socket readiness, register with postmortem, start health checks. */
export async function startDaemon(config: DaemonConfig): Promise<ManagedDaemon>;

/** Probe whether a Unix socket has a live listener (connect + disconnect). */
export async function probeSocket(socketPath: string, timeoutMs?: number): Promise<boolean>;

/** Poll until a socket file appears on disk or deadline is reached. */
export async function waitForSocket(socketPath: string, timeoutMs: number, pollIntervalMs?: number): Promise<void>;
#+end_src

*** Implementation details
1. *Spawn*: =Bun.spawn(config.command, { stdio: ["ignore", "ignore", "pipe"] })= — stderr piped for logging.
2. *No fork*: callers must use =--fg-daemon= (Emacs) or equivalent. =proc= tracks the real PID.
3. *Socket readiness*: =waitForSocket()= polls =fs.access(socketPath)= at 500ms intervals.
4. *Health timer*: =setInterval()= at =config.healthIntervalMs=, =.unref()='d. On socket disappearance, marks dead, calls =config.onCrash()=.
5. *Postmortem*: =postmortem.register(config.name, ...)= on start. The callback: (a) clears health timer, (b) tries sync =proc.kill()= for =Reason.EXIT= (synchronous-only context), (c) tries async =stopCommand= + =terminate()= for signal reasons. Cancel function stored and called on =stop()= to prevent double-dispose.
6. *Graceful stop*: (a) =Bun.$\`${stopCommand}\`.quiet().nothrow()=, (b) wait 2s for =proc.exited=, (c) =terminate({ target: proc })= as escalation, (d) =fs.unlink(socketPath)=.
7. *probeSocket*: extracted from =tryAttachExisting()= — connect to socket, disconnect, return boolean.
8. *stderr*: background async loop reads =proc.stderr=, splits lines, logs via =logger.debug()=.
