* Context
When a user changes a setting (via settings panel or =spell config set=), the change must be persisted back to spell.kdl WITHOUT destroying comments, formatting, or unrelated content. This module implements round-trip KDL editing using =@bgotink/kdl= 0.4.0's format-preserving API.

* Design Decisions
- New file: =packages/coding-agent/src/config/kdl-writer.ts=
- Uses =@bgotink/kdl= =parse()= → mutate → =format()= round-trip (preserves whitespace/comments in parsed representation)
- Uses =KDL_SETTINGS_MAP= to find the correct KDL location for each setting
- Creates missing blocks/nodes on-demand (e.g., if =appearance= block doesn't exist yet, creates it)
- Uses =withFileLock()= from =./file-lock.ts= for concurrent write safety
- Creates parent directories via =Bun.write()= (auto-creates dirs)

** Step 2: Implement =applySettingsToKdl=
#+begin_src typescript
import { type Document, type Node, parse, format } from "@bgotink/kdl";
import { KDL_SETTINGS_MAP, type KdlSettingMapping } from "./kdl-settings-map";
import { getDocumentNode, getChildNode, createStringNode, createBooleanNode, createNumberNode } from "./kdl-helpers";

export function applySettingsToKdl(doc: Document, changes: Map<string, unknown>): Document {
  for (const [settingPath, value] of changes) {
    const mapping = KDL_SETTINGS_MAP[settingPath];
    if (!mapping) continue;

    // Find or create the block node
    let blockNode = getDocumentNode(doc, mapping.block);
    if (!blockNode) {
      blockNode = Node.create(mapping.block);
      blockNode.children = new Document();
      doc.appendNode(blockNode);
    }

    // Navigate to or create the target node within the block
    let parent = blockNode;
    const pathParts = mapping.nodePath.split(".");
    for (let i = 0; i < pathParts.length - 1; i++) {
      let child = getChildNode(parent, pathParts[i]);
      if (!child) {
        child = Node.create(pathParts[i]);
        child.children = new Document();
        if (!parent.children) parent.children = new Document();
        parent.children.appendNode(child);
      }
      parent = child;
    }

    const leafName = pathParts[pathParts.length - 1];
    let targetNode = getChildNode(parent, leafName);

    if (mapping.accessor === "argument") {
      if (!targetNode) {
        // Create new node with value as argument
        targetNode = createNodeForValue(leafName, value);
        if (!parent.children) parent.children = new Document();
        parent.children.appendNode(targetNode);
      } else {
        // Update existing argument
        if (targetNode.hasArgument(mapping.argumentIndex ?? 0)) {
          targetNode.removeArgument(mapping.argumentIndex ?? 0);
        }
        targetNode.addArgument(value as string | number | boolean, undefined, mapping.argumentIndex ?? 0);
      }
    } else {
      // Property accessor
      if (!targetNode) {
        targetNode = Node.create(leafName);
        targetNode.setProperty(mapping.propertyName!, value as string | number | boolean);
        if (!parent.children) parent.children = new Document();
        parent.children.appendNode(targetNode);
      } else {
        targetNode.setProperty(mapping.propertyName!, value as string | number | boolean);
      }
    }
  }
  return doc;
}

** Step 3: Implement =writeKdlSettings=
#+begin_src typescript
import { isEnoent } from "@oh-my-pi/pi-utils";
import { withFileLock } from "./file-lock";

export async function writeKdlSettings(filePath: string, changes: Map<string, unknown>): Promise<void> {
  await withFileLock(filePath, async () => {
    let doc: Document;
    try {
      const content = await Bun.file(filePath).text();
      doc = parse(content);
    } catch (err) {
      if (isEnoent(err)) {
        doc = new Document();
      } else {
        throw err;
      }
    }

    applySettingsToKdl(doc, changes);
    await Bun.write(filePath, format(doc));
  });
}
#+end_src

* Edge Cases
- =Node.create()= produces nodes without formatting metadata → =format()= will use default formatting for new nodes (2-space indent). This is acceptable since new nodes blend with any existing indentation style.
- =clearFormat()= from =@bgotink/kdl= can normalize formatting if needed, but we DON'T call it — we preserve original formatting.
- Property update via =setProperty()= replaces the existing property entry, preserving its position in the entries array.
- Argument update: must remove then add at same index to avoid duplicates.
- Nested node path with missing intermediates (e.g., writing =compaction.enabled= when =model= exists but =compaction= child doesn't) requires creating intermediate nodes with children blocks.
