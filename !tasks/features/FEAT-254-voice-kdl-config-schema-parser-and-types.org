* KDL Syntax

#+begin_src kdl
telegram {
  voice {
    // STT -- incoming voice/audio transcription
    stt-provider "deepgram"           // "deepgram" | "openai"
    stt-api-key "env(DEEPGRAM_API_KEY)"
    stt-model "nova-2"                // provider-specific, optional
    stt-language "en"                 // BCP-47, optional (default "en")

    // TTS -- outgoing voice reply synthesis
    tts-provider "elevenlabs"         // "elevenlabs" | "deepgram"
    tts-api-key "env(ELEVENLABS_API_KEY)"
    tts-model "eleven_multilingual_v2" // provider-specific, optional
    tts-voice "rachel"                // voice ID/name, optional

    // Reply behavior
    reply-mode "mirror"               // "mirror" | "always" | "never"
  }

  user 123456789 {
    voice {
      reply-mode "always"             // per-user override
      tts-voice "adam"                // per-user voice preference
    }
  }
}
#+end_src

** Semantics
- =stt-provider= and =stt-api-key= are required together; if both missing, voice messages get a friendly "Voice not configured" reply
- =tts-provider= and =tts-api-key= are required together; if both missing, reply-mode is forced to ="never"=
- =reply-mode= defaults to ="mirror"= (reply with voice only when user sent voice)
- Per-user =voice {}= overrides merge on top of global: any field set in user overrides the global value
- =env()= support: parse string fields, check for =env(NAME)= or =env(NAME, default=...)= pattern, resolve from =process.env=

* Type Definitions

#+begin_src typescript
type SttProvider = "deepgram" | "openai";
type TtsProvider = "elevenlabs" | "deepgram";
type VoiceReplyMode = "mirror" | "always" | "never";

interface SttConfig {
  provider: SttProvider;
  apiKey: string;
  model?: string;
  language: string; // default "en"
}

interface TtsConfig {
  provider: TtsProvider;
  apiKey: string;
  model?: string;
  voice?: string;
}

interface VoiceConfig {
  stt?: SttConfig;
  tts?: TtsConfig;
  replyMode: VoiceReplyMode; // default "mirror"
}

interface UserVoiceConfig {
  replyMode?: VoiceReplyMode;
  ttsVoice?: string;
}
#+end_src
