#+begin_src typescript
export function getCacheRateTimeSeries(days: number = 30): Array<{ date: string; cacheRate: number; cacheSavings: number }> {
  // Query messages table grouped by day, computing cache rate per day
  const rows = db.prepare(`
    SELECT
      date(timestamp / 1000, 'unixepoch') as date,
      SUM(cache_read_tokens) as total_cache_read,
      SUM(input_tokens) as total_input,
      SUM(cost_cache_read) as cost_cache_read,
      SUM(cost_input) as cost_input
    FROM messages
    WHERE timestamp > ?
    GROUP BY date(timestamp / 1000, 'unixepoch')
    ORDER BY date
  `).all(Date.now() - days * 86400000);

  return rows.map(row => ({
    date: row.date,
    cacheRate: (row.total_input + row.total_cache_read) > 0
      ? row.total_cache_read / (row.total_input + row.total_cache_read)
      : 0,
    cacheSavings: row.cost_input - row.cost_cache_read, // approximate savings
  }));
}
#+end_src
