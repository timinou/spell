const WELL_KNOWN = new Set(['spell-browser', 'spell-browse-mode']);
const TIMESTAMPED_PATTERN = /^spell-browser-\d+-\d+$/;

async function findOrphanedProfiles(options: { registryPath?: string }): Promise<OrphanProfile[]> {
  const registry = new ServiceRegistry(options.registryPath);
  const referenced = new Set((await registry.list()).map(s => s.profileStorage));
  const qtDir = path.join(os.homedir(), '.local', 'share', 'omp-qml-bridge', 'QtWebEngine');
  const entries = await fs.readdir(qtDir).catch(() => []);
  return entries
    .filter(name => !WELL_KNOWN.has(name) && !referenced.has(name))
    .map(name => ({
      name,
      path: path.join(qtDir, name),
      isTimestamped: TIMESTAMPED_PATTERN.test(name),
    }));
}
