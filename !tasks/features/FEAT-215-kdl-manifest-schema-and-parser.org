  setup "readonly-reviewer" {
    domain "coding"
    mode "reviewer"
    tools { allow "read" "grep" "find" }
    sandbox { paths-write "src/" "test/"; bash-allow "bun test*" }
    timeout "30m"
  }
  goal "nightly-tests" {
    setup "readonly-reviewer"
    schedule type="cron" expression="0 2 * * *" jitter="5m"
    prompt "Run the full test suite."
    state persist=true
    hooks {
      on-success { webhook "https://hooks.slack.com/..." }
      on-failure { telegram chat-id=12345 }
    }
  }
