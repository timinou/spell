// packages/coding-agent/src/config/project-detection.ts
type ProjectLanguage = "typescript" | "javascript" | "rust" | "python" | "go" | "ruby" | "jvm" | "unknown";

interface DetectedToolchain {
    testCmd?: string;    // "bun test", "cargo test", "pytest"
    checkCmd?: string;   // "tsgo", "cargo clippy", "mypy"
    lintCmd?: string;    // "biome check", "eslint", "ruff"
    formatCmd?: string;  // "biome format", "cargo fmt", "black"
}

interface DetectedProject {
    name: string;
    language: ProjectLanguage;
    frameworks: string[];
    toolchain: DetectedToolchain;
    packageManager?: string; // "bun", "npm", "yarn", "cargo", "pip"
    monorepo: boolean;
}

export async function detectProject(cwd: string): Promise<DetectedProject>;

* Detection Strategy (coding domain only)

** TypeScript/JavaScript
Read =package.json=:
- =scripts.test= → testCmd (look for bun test, jest, vitest, mocha)
- =scripts.check= or =scripts.typecheck= → checkCmd
- =scripts.lint= → lintCmd
- Detect package manager from lockfile: =bun.lock= → bun, =package-lock.json= → npm, =yarn.lock= → yarn, =pnpm-lock.yaml= → pnpm
- Framework detection from deps: next, react, vue, svelte, express, fastify, hono
- Monorepo: =workspaces= field in package.json
- Type checker: presence of =tsgo.json= → tsgo, =tsconfig.json= → tsc
- Linter: =biome.json= → biome, =eslint.config.*= or =.eslintrc.*= → eslint

** Rust
- =Cargo.toml= exists → testCmd = "cargo test", checkCmd = "cargo clippy", formatCmd = "cargo fmt"
- =workspace.members= in Cargo.toml → monorepo

** Python
- =pyproject.toml= or =setup.py= → testCmd from =tool.pytest= section or default "pytest"
- =tool.mypy= → checkCmd = "mypy"
- =tool.ruff= → lintCmd = "ruff check", formatCmd = "ruff format"

** Go
- =go.mod= exists → testCmd = "go test ./...", checkCmd = "go vet ./..."
