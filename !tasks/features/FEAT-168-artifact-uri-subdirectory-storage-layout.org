* Initial Message

make artifact:// have an extension and show it everywhere it is mentioned, use artifact:// for images and screenshot tools, and use the artifact:// uri system for artiface recording during plan executions (explain that)

* Context

Change artifact:// URI format from =artifact://{id}= to =artifact://{toolType}/{id}.{ext}=.
Change on-disk storage from flat ={id}.{toolType}.log= to subdirectories ={toolType}/{id}.{ext}=.

URI examples:
- =artifact://bash/0.txt= → =artifacts/bash/0.txt=
- =artifact://screenshot/3.png= → =artifacts/screenshot/3.png=
- =artifact://generate_image/5.png= → =artifacts/generate_image/5.png=
- =artifact://fetch/2.txt= → =artifacts/fetch/2.txt=
- =artifact://render_mermaid/1.txt= → =artifacts/render_mermaid/1.txt=

Backward compatibility: old =artifact://{id}= URIs (numeric-only host, no path) fall back
to scanning the root artifacts dir for ={id}.*= files (existing behavior).

** Decision: Extension mapping

Default extension is =txt=. Tool-specific overrides:
- =screenshot= → =png=
- =generate_image= → =png= (or mime-derived at save time)
- All text tool outputs (bash, python, ssh, fetch, grep, find, read, render_mermaid, async) → =txt=

** Decision: ID allocation

Global monotonic counter remains. IDs are unique across all tool types.
=allocatePath= signature changes to accept extension parameter.

** Decision: URI parsing

For URI =artifact://bash/5.txt=:
- =rawHost= (from =InternalUrlRouter= URL parsing) = =bash= (tool type)
- =pathname= = =/5.txt= (id + extension)
- Resolved to ={artifactsDir}/bash/5.txt=

For legacy URI =artifact://5= (backward compat):
- =rawHost= is numeric → scan root for ={id}.*= files (existing behavior)

   async resolve(url: InternalUrl): Promise<InternalResource> {
     const artifactsDir = this.options.getArtifactsDir();
     if (!artifactsDir) throw new Error("No session - artifacts unavailable");

     const host = url.rawHost || url.hostname;
     if (!host) throw new Error("artifact:// URL requires an ID");

     // New format: artifact://toolType/id.ext
     if (!/^\d+$/.test(host)) {
       const pathname = (url.rawPathname || url.pathname).replace(/^\//, "");
       if (!pathname) throw new Error(`artifact:// URL missing filename: artifact://${host}/...`);
       const relPath = `${host}/${pathname}`;
       const filePath = path.join(artifactsDir, relPath);
       // Security: prevent path traversal
       if (!filePath.startsWith(artifactsDir)) {
         throw new Error("artifact:// path traversal denied");
       }
       try {
         const content = await Bun.file(filePath).text();
         return {
           url: url.href,
           content,
           contentType: contentTypeFromExt(path.extname(filePath)),
           size: Buffer.byteLength(content, "utf-8"),
           sourcePath: filePath,
         };
       } catch (err) {
         if (isEnoent(err)) {
           const available = await listAvailableArtifacts(artifactsDir);
           throw new Error(`Artifact ${relPath} not found. Available: ${available.join(", ") || "none"}`);
         }
         throw err;
       }
     }

     // Legacy format: artifact://id (numeric)
     // ... existing scan-root behavior ...
   }

    const mgr = new ArtifactManager(sessionFile);
    const r1 = await mgr.allocatePath("bash", "txt");

    const r3 = await mgr.allocatePath("fetch", "txt");

    expect(r2.id).toBe("1");

import { afterEach, beforeEach, describe, expect, it } from "bun:test";
import * as fs from "node:fs/promises";
import * as os from "node:os";
import * as path from "node:path";
import { ArtifactProtocolHandler } from "../../src/internal-urls/artifact-protocol";
import { InternalUrlRouter } from "../../src/internal-urls/router";

describe("ArtifactProtocolHandler", () => {
  let tmpDir: string;
  let artifactsDir: string;
  let handler: ArtifactProtocolHandler;
  let router: InternalUrlRouter;

  beforeEach(async () => {
    tmpDir = await fs.mkdtemp(path.join(os.tmpdir(), "artifact-proto-test-"));
    artifactsDir = path.join(tmpDir, "artifacts");
    await fs.mkdir(artifactsDir, { recursive: true });
    handler = new ArtifactProtocolHandler({ getArtifactsDir: () => artifactsDir });
    router = new InternalUrlRouter();
    router.register(handler);
  });

  afterEach(async () => {
    await fs.rm(tmpDir, { recursive: true, force: true });
  });

  it("resolves new format artifact://bash/5.txt", async () => {
    await fs.mkdir(path.join(artifactsDir, "bash"), { recursive: true });
    await Bun.write(path.join(artifactsDir, "bash", "5.txt"), "hello");
    const result = await router.resolve("artifact://bash/5.txt");
    expect(result.content).toBe("hello");
    expect(result.sourcePath).toBe(path.join(artifactsDir, "bash", "5.txt"));
  });

  it("resolves legacy format artifact://5", async () => {
    await Bun.write(path.join(artifactsDir, "5.bash.log"), "legacy");
    const result = await router.resolve("artifact://5");
    expect(result.content).toBe("legacy");
  });

  it("returns text/plain for .txt", async () => {
    await fs.mkdir(path.join(artifactsDir, "bash"), { recursive: true });
    await Bun.write(path.join(artifactsDir, "bash", "0.txt"), "data");
    const result = await router.resolve("artifact://bash/0.txt");
    expect(result.contentType).toBe("text/plain");
  });

  it("returns sourcePath for image artifacts", async () => {
    await fs.mkdir(path.join(artifactsDir, "screenshot"), { recursive: true });
    // Write a minimal PNG (just need file to exist for sourcePath)
    await Bun.write(path.join(artifactsDir, "screenshot", "1.png"), "fake-png");
    const result = await router.resolve("artifact://screenshot/1.png");
    expect(result.sourcePath).toBe(path.join(artifactsDir, "screenshot", "1.png"));
  });

  it("throws for non-existent new-format artifact", async () => {
    await expect(router.resolve("artifact://bash/999.txt")).rejects.toThrow(/not found/);
  });

  it("throws for non-existent legacy artifact", async () => {
    await expect(router.resolve("artifact://999")).rejects.toThrow(/not found/);
  });

  it("prevents path traversal", async () => {
    await expect(router.resolve("artifact://../etc/0.txt")).rejects.toThrow(/traversal/);
  });
});
