Change artifact:// URI format from =artifact://{id}= to:

=artifact://<session-id>/<agent-name>/<tool>/<number>.<ext>=

Where:
- =<session-id>= is the Snowflake hex ID of the ROOT session (e.g., =14b64b9276f08680=)
- =<agent-name>= is =main= for the main agent, or the task prefix for subagents (e.g., =0-FindArtifactFiles=)
- =<tool>= is the tool type (=bash=, =screenshot=, =generate_image=, etc.)
- =<number>.<ext>= is sequential ID + file extension (=0.txt=, =1.png=, etc.)

URI examples:
- =artifact://14b64b9276f08680/main/bash/0.txt=
- =artifact://14b64b9276f08680/main/screenshot/3.png=
- =artifact://14b64b9276f08680/0-FindArtifactFiles/bash/0.txt=
- =artifact://14b64b9276f08680/1-UpdateCallers/fetch/2.txt=

On-disk, each agent stores artifacts in its own session's artifact dir as ={tool}/{number}.{ext}=.

Backward compatibility: old =artifact://{id}= URIs (numeric-only host, no path) fall back
to scanning current session's root artifacts dir for ={id}.*= files.

** Decision: Extension mapping

Default extension is =txt=. Tool-specific overrides:
- =screenshot= → =png=
- =generate_image= → =png= (or mime-derived at save time)
- All text tool outputs (bash, python, ssh, fetch, grep, find, read, render_mermaid, async) → =txt=

** Decision: ID allocation

Global monotonic counter per ArtifactManager instance. IDs are unique within an agent's
scope (main or subagent), not globally across the session tree.

** Decision: Session/agent identity

- Main agent: =rootSessionId= = own session ID, =agentName= = ="main"=
- Subagent: =rootSessionId= = parent's session ID (from =parentSession= header field or
  passed via task executor), =agentName= = =parentTaskPrefix= (e.g., ="0-FindArtifactFiles"=)

The =ArtifactManager= constructor needs these two new parameters.

** Decision: URI parsing

URL =artifact://14b64b9276f08680/main/bash/0.txt= parses as:
- =rawHost= = =14b64b9276f08680= (session ID)
- =pathname= = =/main/bash/0.txt=
- Protocol handler splits pathname: =agentName=main=, =tool=bash=, =filename=0.txt=

Legacy URL =artifact://5=:
- =rawHost= = =5= (numeric) → legacy fallback in current session

** Decision: Cross-session resolution

Sessions root: =~/.spell/agent/sessions/= contains per-project subdirs.
Resolver scans all project subdirs for a session file matching =*_{sessionId}.jsonl=.
Artifact dir = session file path minus =.jsonl= extension.

For the current session + current agent: direct lookup (no scan).
For subagents of current session: session manager tracks spawned session IDs + paths.
For other sessions: filesystem scan across session repos.

The resolver result SHOULD be cached for the lifetime of a protocol handler instance.

#+begin_src typescript
async resolve(url: InternalUrl): Promise<InternalResource> {
  const host = url.rawHost || url.hostname;
  if (!host) throw new Error("artifact:// URL requires a session ID or numeric ID");

  // Legacy format: artifact://5 (numeric host)
  if (/^\d+$/.test(host)) {
    return this.#resolveLegacy(host);
  }

  // New format: artifact://sessionId/agentName/tool/number.ext
  const pathParts = (url.rawPathname || url.pathname)
    .replace(/^\//, "")
    .split("/");

  if (pathParts.length < 3) {
    throw new Error(
      `artifact:// URL must be artifact://<session-id>/<agent>/<tool>/<file>. Got: ${url.href}`
    );
  }

  const [agentName, tool, ...fileParts] = pathParts;
  const filename = fileParts.join("/");
  const sessionId = host;

  // Resolve artifact directory for this session + agent
  const artifactsDir = await this.#resolveArtifactsDir(sessionId, agentName);
  if (!artifactsDir) {
    throw new Error(`Cannot resolve session ${sessionId} for artifact lookup`);
  }

  const filePath = path.join(artifactsDir, tool, filename);
  // Security: prevent path traversal
  if (!filePath.startsWith(artifactsDir)) {
    throw new Error("artifact:// path traversal denied");
  }

  try {
    const content = await Bun.file(filePath).text();
    return {
      url: url.href,
      content,
      contentType: contentTypeFromExt(path.extname(filePath)),
      size: Buffer.byteLength(content, "utf-8"),
      sourcePath: filePath,
    };
  } catch (err) {
    if (isEnoent(err)) {
      const available = await listAvailableArtifacts(artifactsDir);
      throw new Error(`Artifact not found: ${tool}/${filename}. Available: ${available.join(", ") || "none"}`);
    }
    throw err;
  }
}

#+begin_src typescript
async #resolveArtifactsDir(sessionId: string, agentName: string): Promise<string | null> {
  const currentSessionId = this.options.getSessionId();
  const currentAgentName = this.options.getAgentName();

  // Current session + current agent → direct
  if (sessionId === currentSessionId && agentName === currentAgentName) {
    return this.options.getArtifactsDir();
  }

  // Spawned subagent of current session → ask session manager
  if (sessionId === currentSessionId && this.options.resolveSpawnedSessionDir) {
    // agentName is the task prefix; session manager maps it to a session dir
    const dir = this.options.resolveSpawnedSessionDir(agentName);
    if (dir) return dir;
  }

  // Cross-session: scan sessions root
  const sessionsRoot = this.options.getSessionsRoot();
  if (!sessionsRoot) return null;
  return this.#scanForSession(sessionsRoot, sessionId);
}

async #scanForSession(sessionsRoot: string, sessionId: string): Promise<string | null> {
  if (this.#sessionDirCache.has(sessionId)) return this.#sessionDirCache.get(sessionId)!;

  // Scan all project subdirectories
  const repoDirs = await fs.readdir(sessionsRoot, { withFileTypes: true });
  for (const repo of repoDirs) {
    if (!repo.isDirectory()) continue;
    const repoPath = path.join(sessionsRoot, repo.name);
    const files = await fs.readdir(repoPath);
    const match = files.find(f => f.endsWith(`_${sessionId}.jsonl`));
    if (match) {
      const artifactsDir = path.join(repoPath, match.slice(0, -6)); // remove .jsonl
      this.#sessionDirCache.set(sessionId, artifactsDir);
      return artifactsDir;
    }
  }
  return null;
}

async function listAvailableArtifacts(artifactsDir: string): Promise<string[]> {
  try {
    const result: string[] = [];
    const entries = await fs.readdir(artifactsDir, { withFileTypes: true });
    for (const entry of entries) {
      if (entry.isDirectory()) {
        const subFiles = await fs.readdir(path.join(artifactsDir, entry.name));
        for (const f of subFiles) {
          if (/^\d+\./.test(f)) result.push(`${entry.name}/${f}`);

        result.push(entry.name.split(".")[0]); // legacy bare ID
      }
    }
    return result.sort();
  } catch { return []; }
}

describe("ArtifactProtocolHandler", () => {
  let tmpDir: string;

    tmpDir = await fs.mkdtemp(path.join(os.tmpdir(), "artifact-proto-test-"));
    sessionsRoot = path.join(tmpDir, "sessions");
    const projectDir = path.join(sessionsRoot, "-code-project");
    await fs.mkdir(projectDir, { recursive: true });

    const sessionFile = path.join(projectDir, "2026-01-01T00-00-00-000Z_abc123.jsonl");
    await Bun.write(sessionFile, "");
    artifactsDir = path.join(projectDir, "2026-01-01T00-00-00-000Z_abc123");
    await fs.mkdir(artifactsDir, { recursive: true });

    const handler = new ArtifactProtocolHandler({
      getArtifactsDir: () => artifactsDir,
      getSessionId: () => "abc123",
      getAgentName: () => "main",
      getSessionsRoot: () => sessionsRoot,
    });
    router = new InternalUrlRouter();
    router.register(handler);
  });

  afterEach(async () => {
    await fs.rm(tmpDir, { recursive: true, force: true });
  });

  it("resolves current session main agent artifact", async () => {
    await fs.mkdir(path.join(artifactsDir, "bash"), { recursive: true });
    await Bun.write(path.join(artifactsDir, "bash", "5.txt"), "hello");
    const result = await router.resolve("artifact://abc123/main/bash/5.txt");
    expect(result.content).toBe("hello");
    expect(result.sourcePath).toBe(path.join(artifactsDir, "bash", "5.txt"));
  });

    // Create another session in a different project repo

    expect(result.content).toBe("cross-session");
