Fixed code adds storage dir cleanup:
#+begin_src typescript
async teardown(options?: { preserveStorage?: boolean }): Promise<void> {
  try {
    await this.#journey.teardown();
  } finally {
    await fs.rm(this.#settingsFile, { force: true });
    if (!options?.preserveStorage && this.#storageName) {
      const storageDir = path.join(
        os.homedir(), '.local', 'share', 'omp-qml-bridge', 'QtWebEngine', this.#storageName
      );
      await fs.rm(storageDir, { recursive: true, force: true });
    }
  }
}
#+end_src

Note: =preserveStorage: true= is used in cookie-persistence test between first and second launch.

This requires storing =storageName= in BrowserJourney (currently only stored in props passed to QML). The =launch= method already receives =storageName= in options — store it as =#storageName= field.
