#+begin_src typescript
import type { Document } from "@bgotink/kdl";
import { getDocumentNode, getStringArguments, getStringArgument } from "./kdl-helpers";

export function parseKeybindingsBlock(doc: Document): Record<string, string | string[]> | undefined {
  const node = getDocumentNode(doc, "keybindings");
  if (!node?.children) return undefined;
  
  const config: Record<string, string | string[]> = {};
  for (const child of node.children.nodes) {
    const action = child.getName();
    const keys = getStringArguments(child);
    if (keys.length === 1) {
      config[action] = keys[0];
    } else if (keys.length > 1) {
      config[action] = keys;
    }
  }
  return config;
}
#+end_src

The =parseKdlFrontmatterContent= helper parses the KDL fragment. For MODE.md, it expects a single =mode= node with children. It flattens the mode's children into a flat =Record<string, unknown>=, matching the shape YAML frontmatter produces.

Import =parse= from =@bgotink/kdl= in frontmatter.ts (new dependency for this file).

* Edge Cases
- Keybindings: KDL node names use kebab-case (=toggle-plan=) while KeybindingsConfig uses camelCase (=togglePlan=). Convert kebab-case → camelCase during parsing.
- Modes: =tools= child can be a node with allow/deny children, or absent (no tool restrictions). Both shapes must be handled.
- Frontmatter: =normalizeKeys= already converts kebab-case to camelCase (=read-only= → =readOnly=). Verify this works for KDL-produced keys too.
- MODE.md with KDL frontmatter must produce the SAME ModeConfigFrontmatter as YAML frontmatter for the same logical config. Downstream consumers (mode-helpers.ts resolveModeConfig) must not need changes.
