#+begin_src typescript
const snapshot = this.#kernel.getState(loopId);
if (snapshot.manifest && options.completedTickets?.length) {
  const manifest = snapshot.manifest;
  for (const ticketId of options.completedTickets) {
    try {
      this.#ticketLifecycle.completeTicket(manifest, ticketId, snapshot.iteration);
    } catch (err) {
      logger.warn("Failed to complete ticket", { loopId, ticketId, error: String(err) });
    }
  }
  for (const ticketId of options.activeTickets ?? []) {
    try {
      this.#ticketLifecycle.startTicket(manifest, ticketId, snapshot.iteration);
    } catch (err) {
      logger.warn("Failed to start ticket", { loopId, ticketId, error: String(err) });
    }
  }
  // Set updated manifest on snapshot
  this.#kernel.updateLoop(loopId, loop => { loop.manifest = manifest; }, "manifest.tickets_updated", {
    completed: options.completedTickets,
    active: options.activeTickets,
  });
}
#+end_src

** Step 3: Fire ticket gates
After ticket transitions, for each completed ticket, evaluate gates with =ticketCompleted=:
#+begin_src typescript
if (snapshot.manifest && options.completedTickets?.length) {

      state: snapshot.state,

    }, { cwd: this.#cwd, attemptNumber: 1, evidence: [] });

}
#+end_src
