  systemPrompt: SystemPrompt | undefined,
  options: SystemBlockOptions = {},

  const { includeClaudeCodeInstruction = false, extraInstructions = [], billingPayload, cacheControl } = options;

  const blocks: AnthropicSystemBlock[] = [];

  // Claude Code billing header (uncached — must come first for billing)
  const promptBlocks = systemPromptBlocks(systemPrompt);
  const fullText = systemPromptText(systemPrompt);
  const sanitizedFullText = fullText ? fullText.toWellFormed() : "";
  const hasBillingHeader = sanitizedFullText.includes(CLAUDE_BILLING_HEADER_PREFIX);

  if (includeClaudeCodeInstruction && !hasBillingHeader) {
    const payloadSeed = billingPayload ?? { system: sanitizedFullText, extraInstructions: extraInstructions.map(i => i.trim()).filter(Boolean) };
    blocks.push(
      { type: "text", text: createClaudeBillingHeader(payloadSeed) },
      { type: "text", text: claudeCodeSystemInstruction },
    );
  }

  // Extra instructions (cached)
  for (const instruction of extraInstructions.map(i => i.trim()).filter(Boolean)) {
    blocks.push({
      type: "text", text: instruction,
      ...(cacheControl ? { cache_control: cacheControl } : {}),
    });
  }

  // System prompt blocks — respect stable/dynamic distinction
  if (promptBlocks.length > 0) {
    for (const block of promptBlocks) {
      const isStable = block.stable !== false; // default true
      blocks.push({
        type: "text",
        text: block.text.toWellFormed(),
        // Only apply cache_control to stable blocks
        ...(isStable && cacheControl ? { cache_control: cacheControl } : {}),
      });
    }
  }

  return blocks.length > 0 ? blocks : undefined;
}

*** =applyPromptCaching= (line ~1049) — IMPORTANT FIX

The function currently unconditionally adds =cache_control= to the last system block (around line 1069-1072):
#+begin_src typescript
if (params.system && Array.isArray(params.system) && params.system.length > 0) {
  applyCacheControlToLastBlock(params.system, cacheControl);
  cacheBreakpointsUsed++;
}
#+end_src

This is WRONG when multi-block caching is active because the last system block is the DYNAMIC block (=stable: false=) — we don't want it cached.

**Fix**: Skip the system block step when any system block already has =cache_control= (indicating =buildAnthropicSystemBlocks= already placed breakpoints on stable blocks):

#+begin_src typescript
if (params.system && Array.isArray(params.system) && params.system.length > 0) {
  // Skip if stable blocks already have cache_control from buildAnthropicSystemBlocks
  const hasExistingCacheControl = params.system.some(
    (block: AnthropicSystemBlock & CacheControlBlock) => block.cache_control != null
  );
  if (!hasExistingCacheControl) {
    applyCacheControlToLastBlock(params.system, cacheControl);
  }
  // Count existing cache_control breakpoints in system blocks toward the limit
  const systemCachePoints = params.system.filter(
    (block: AnthropicSystemBlock & CacheControlBlock) => block.cache_control != null
  ).length;
  cacheBreakpointsUsed += systemCachePoints || 1; // at least 1 if we just added
}
#+end_src

This ensures:
- When =SystemPromptBlock[]= with stable/dynamic split is used: =buildAnthropicSystemBlocks= places =cache_control= on stable blocks, =applyPromptCaching= detects them and skips the system step
- When plain string is used (backward compat): no pre-existing =cache_control=, so =applyPromptCaching= adds one to the last block as before

#+begin_src typescript
function buildSystemPrompt(
  systemPrompt: SystemPrompt | undefined,
  model: Model<"bedrock-converse-stream">,
  cacheRetention: CacheRetention,
): SystemContentBlock[] | undefined {
  const blocks = systemPromptBlocks(systemPrompt);
  if (blocks.length === 0) return undefined;

  const result: SystemContentBlock[] = [];
  const supportsCache = cacheRetention !== "none" && supportsPromptCaching(model);

  for (const block of blocks) {
    result.push({ text: block.text.toWellFormed() });
    // Add cache point only after stable blocks
    if (supportsCache && block.stable !== false) {
      result.push({
        cachePoint: { type: CachePointType.DEFAULT, ...(cacheRetention === "long" ? { ttl: CacheTTL.ONE_HOUR } : {}) },
      });
    }
  }

  return result;
}
#+end_src
