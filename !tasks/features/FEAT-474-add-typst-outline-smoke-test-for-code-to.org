#+begin_src typescript
it("formats Typst outline result from native code buffer", async () => {
    const typstOutline = [
        { name: "\"theme.typ\"", kind: "import", line: 1, end_line: 1, column: 1, exported: false, signature: "import \"theme.typ\": *" },
        { name: "title", kind: "let", line: 2, end_line: 2, column: 1, exported: false, signature: "let title =" },
        { name: "heading.where(level: 1)", kind: "show", line: 3, end_line: 3, column: 1, exported: false, signature: "show heading.where(level: 1):" },
    ];
    const bufferSpy = spyOn(nativesModule, "executeCodeBuffer").mockReturnValue({
        output: typstOutline,
        error: false,
    });
    const tool = new CodeTool(createSession());
    const result = await tool.execute("tool", { command: "outline", file: "/tmp/test/doc.typ" });
    const text = result.content.find(c => c.type === "text")?.text ?? "";
    const parsed = JSON.parse(text);

    expect(parsed).toHaveLength(3);
    expect(parsed[0].kind).toBe("import");
    expect(parsed[1].kind).toBe("let");
    expect(parsed[2].kind).toBe("show");
    expect(parsed[2].name).toBe("heading.where(level: 1)");

    expect(bufferSpy).toHaveBeenCalledWith(
        expect.objectContaining({
            command: "outline",
            file: "/tmp/test/doc.typ",
        }),
    );
});
#+end_src

** What this proves

- The code tool correctly passes the outline command and file path to =executeCodeBuffer=
- The result is correctly JSON-serialized and returned as text content
- The Typst-specific declaration kinds (import, let, show) survive the formatting pipeline
- A future regression in the wiring will fail this test before hitting production
