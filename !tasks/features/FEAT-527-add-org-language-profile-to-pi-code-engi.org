* Tree-sitter-org AST node types (from existing =pi-org-engine= usage)

Key node hierarchy:
- =document= → top-level root
  - =section= → contains headline + body + sub-sections
    - =headline= → contains =stars=, =item= (title + optional TODO keyword), =tag_list=
    - =property_drawer= → contains =property= nodes
    - =body= → free-form content
    - =section= → nested sub-sections

* Language profile design

#+BEGIN_SRC rust
fn org_profile() -> LanguageProfile {
    let gd = generated::org::grammar();
    LanguageProfile {
        id: LanguageId::new(\"org\"),
        extensions: vec![\"org\".into()],
        declarations: vec![
            // Headings are the primary declarations
            DeclarationPattern {
                node_types: vec![\"section\".into()],
                name_field: \"headline\".into(),  // or first child
                kind: \"heading\".into(),
                body_field: Some(\"body\".into()),
                visibility: None,
            },
        ],
        class_like: vec![
            // Sections contain sub-sections (headings contain sub-headings)
            ClassLikePattern {
                node_type: \"section\".into(),
                body_field: \"body\".into(),
                member_types: vec![\"section\".into()],
            },
        ],
        imports: vec![],
        exports: vec![],
        references: vec![],
        separators: vec![],
        production_rules: gd.production_rules,
        inverse_rules: gd.inverse_rules,
        all_types: gd.all_types,
        supertypes: gd.supertypes,
        ts_language: tree_sitter_org::LANGUAGE.into(),
    }
}
#+END_SRC
