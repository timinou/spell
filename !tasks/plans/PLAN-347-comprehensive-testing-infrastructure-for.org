  failing_tool = %{
    kind: :ptc,
    name: "exploder",
    source: "(fail \"boom\")"
  }
  
  ToolRegistry.put(failing_tool)
  
  {:ok, session} = start_session()
  result = Session.run(session, "use the exploder tool")
  
  # Agent sees error, adapts
  assert {:ok, recovery} = result
  assert recovery =~ "tool failed"

    "define a counter tool",
    "increment the counter",  
    "show counter value",
    "reset counter"
  ]

```elixir
test "parallel telemetry doesn't corrupt store" do
  {:ok, store} = Store.start_link()
  
  tasks = for i <- 1..100 do
    Task.async(fn ->
      emit_span_lifecycle(store, "span-#{i}")
    end)
  end
  
  Task.await_many(tasks)
  
  state = :sys.get_state(store)
  assert map_size(state.spans) == 100
  assert forest_valid?(state.spans)
end

test "subscriber cleanup on crash" do
  {:ok, store} = Store.start_link()
  
  crasher = spawn_link(fn ->
    Store.subscribe(store)
    raise "boom"
  end)
  
  Process.sleep(10)
  
  state = :sys.get_state(store)
  assert crasher not in state.subscribers
end
```

## Testing Philosophy

### Behavioral First
Test what the system DOES, not how it does it. A refactor should not break tests if behavior is preserved.

### Property-Driven
Use generators to explore the state space. Find edge cases the developer didn't consider.

### Visual Confidence  
See what the user sees. Catch rendering regressions before they ship.

### Fast Feedback
Tests run in milliseconds. Developers run them continuously.

### Production-Like
Test in conditions as close to production as possible. Use real supervisors, real telemetry.

### Behavioral Scenario Test ✓
```elixir
feature "tree navigation" do
  scenario "expanding a run shows its tools" do
    given "a collapsed run with 3 tools" do
      forest = build_forest(run: [tool1, tool2, tool3])
      runner = start_app(initial: forest)
    end
    
    when "user presses space on the run" do
      runner
      |> key("space")
      |> wait_for_render()
    end
    
    then "all 3 tools are visible" do
      assert visible_rows(runner) =~ "tool1"
      assert visible_rows(runner) =~ "tool2"  
      assert visible_rows(runner) =~ "tool3"
    end
  end
end
```

### Property Hypothesis Test ✓

property "tool composition preserves types" do
