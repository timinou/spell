:PROPERTIES:
:CUSTOM_ID: PROJ-055-code-tool-surface-cutover-3-command-mode::proj-new-schema
:END:
- File: =packages/coding-agent/src/tools/code.ts=
- Replace =codeSchema= with discriminated union on =command: "get" | "edit" | "manage"=
