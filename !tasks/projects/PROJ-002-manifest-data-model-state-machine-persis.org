interface ManifestTicket {
  id: string;              // CUSTOM_ID from spec (e.g., FEAT-001-auth-api)
  title: string;
  state: OrgTodoState;     // ITEM | DOING | DONE | BLOCKED | HOLD
  specPath?: string;       // Source spec file path
  orgItemId?: string;      // Pre-existing org item ID (if from orgItemIds source)
  acceptanceCriteria: string[];
  dependencies: string[];  // BLOCKER IDs
  triggers: string[];      // TRIGGER expressions
  gates: LoopGateConfig[]; // Derived gates for this ticket
  effort?: string;         // EFFORT property
  priority?: string;       // PRIORITY property
  layer?: string;          // LAYER property
  tags: string[];
  changedFiles: string[];  // Files changed while working this ticket
  findings: string[];      // Open findings for this ticket
  childLoopId?: string;    // If decomposed into child loop
  iterationHistory: number[]; // Which iterations worked on this ticket
}

interface ManifestSnapshot {
  version: number;         // Incremented on replan
  tickets: ManifestTicket[];
  dependencyEdges: Array<{ from: string; to: string }>; // Derived from BLOCKER
  triggerRules: Array<{ source: string; target: string; keyword: string }>; // From TRIGGER
  manifestOrgPath: string; // Path to manifest.org
  createdAt: number;
  updatedAt: number;
}

* Edge Cases
- Manifest with circular dependencies (BLOCKER cycles): detect and reject during manifest creation
- Empty manifest (no tickets): reject at loop_launch, not at manifest creation
- Ticket with CUSTOM_ID collision: reject during manifest building
- Org file parse failure: surface error with file path and line number
- Checkpoint with manifest but missing ticket files: reconstruct from event log
